\begin{figure}[t]
\centering
\includegraphics[width=\columnwidth]{fig_har_convergence.png}
\caption{Central test accuracy against communication round on UCI-HAR over
100 rounds, means of three seeds with $\pm 1$ SD bands. No stopping marks are
drawn because none is resolvable here: the largest peak-to-final drop is
$0.032$ at $\varepsilon = 1$ against a $\pm 0.042$ across-seed spread. What
the figure does show is the quality ceiling --- the non-private model reaches
$0.88$ while $\varepsilon = 0.5$ plateaus near $0.47$, and no amount of
further training closes that gap, because the training itself spends budget.}
\label{fig:convergence}
\end{figure}

\begin{figure}[t]
\centering
\includegraphics[width=\columnwidth]{fig_cifar_stopping.png}
\caption{The same curves on CIFAR-10 over 30 rounds, where a stopping round is
resolvable under a tight budget. Markers show the energy-optimal stopping
round, drawn only where the peak-to-final drop exceeds the across-seed spread
of final accuracy: at $\varepsilon = 0.5$ (round 8, all three seeds peaking
within one round of each other) and $\varepsilon = 1$ (rounds 10--16). At
$\varepsilon \ge 2$ and without privacy no decline clears the spread, so no
marker is drawn. The rounds after the $\varepsilon = 0.5$ peak account for
$60$--$68\%$ of the run's net energy and buy negative accuracy.}
\label{fig:stopping}
\end{figure}
